<!-- YOUTUBE PLAYER API -->

<script>
let player;
let playerReady = false;
let musicStarted = false;

function onYouTubeIframeAPIReady() {

    player = new YT.Player('player', {

        height: '200',
        width: '200',

        videoId: 'omFa4Yk9Fmg',

        playerVars: {
            autoplay: 0,
            controls: 1,
            playsinline: 1,
            rel: 0
        },

        events: {

            onReady: function() {
                playerReady = true;
            }

        }

    });
}


// Start birthday website

function startBirthday() {

    document.getElementById('welcome').style.display = 'none';

    document.getElementById('birthday').style.display = 'block';

    window.scrollTo(0, 0);

    musicStarted = true;

    if (playerReady) {
        player.unMute();
        player.setVolume(70);
        player.playVideo();
    }

    document.getElementById('musicButton').innerText =
        '♫ MUTE MUSIC';

    startHearts();
}


// Music toggle

function toggleMusic() {

    if (!playerReady) {
        alert("Music is still loading. Please try again!");
        return;
    }

    if (player.isMuted()) {

        player.unMute();
        player.playVideo();

        document.getElementById('musicButton').innerText =
            '♫ MUTE MUSIC';

    } else {

        player.mute();

        document.getElementById('musicButton').innerText =
            '♫ UNMUTE MUSIC';
    }
}


// Gift button

function openGift() {

    let letter = document.getElementById('letter');
    let button = document.getElementById('giftButton');

    if (letter.style.display === 'block') {

        letter.style.display = 'none';
        button.innerText = 'OPEN YOUR GIFT 🎀';

    } else {

        letter.style.display = 'block';
        button.innerText = 'CLOSE YOUR LETTER ♡';

        letter.scrollIntoView({
            behavior: 'smooth',
            block: 'start'
        });

        startHearts();
    }
}


// Floating hearts

function startHearts() {

    for (let i = 0; i < 8; i++) {

        setTimeout(() => {

            let heart = document.createElement('div');

            heart.className = 'heart';
            heart.innerHTML = '♥';

            heart.style.left =
                Math.random() * 100 + 'vw';

            heart.style.fontSize =
                (15 + Math.random() * 25) + 'px';

            document.body.appendChild(heart);

            setTimeout(() => {
                heart.remove();
            }, 7000);

        }, i * 250);

    }

}
